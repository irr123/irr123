{{/* layouts/_default/_markup/render-image.html */}} {{- $src := .Destination -}} {{- $alt := .PlainText -}} {{- $title := .Title -}} {{- $image := "" -}} {{- with .Page.Resources.GetMatch $src -}} {{- $image = . -}} {{- else -}} {{- with resources.Get $src -}} {{- $image = . -}} {{- end -}} {{- end -}} {{- $isStatic := false -}} {{- if not $image -}} {{- $staticCheck := printf "static%s" $src -}} {{- if fileExists $staticCheck -}} {{- $isStatic = true -}} {{- end -}} {{- end -}} {{- if $image -}} {{- $originalPermalink := $image.RelPermalink -}} {{- if eq $image.MediaType.SubType "svg" -}} {{- else -}} {{- $jpeg := $image.Process "jpeg q85" -}} {{- $webpSet := slice -}} {{- $jpegSet := slice -}} {{- range $w := (slice 360 540 720 1080 1440) -}} {{- if lt $w $image.Width -}} {{- $rw := $image.Process (printf "resize %dx webp q75" $w) -}} {{- $webpSet = $webpSet | append (printf "%s %dw" $rw.RelPermalink $w) -}} {{- $rj := $image.Process (printf "resize %dx jpeg q85" $w) -}} {{- $jpegSet = $jpegSet | append (printf "%s %dw" $rj.RelPermalink $w) -}} {{- end -}} {{- end -}} {{- $webp := $image.Process "webp q75" -}} {{- $webpSet = $webpSet | append (printf "%s %dw" $webp.RelPermalink $image.Width) -}} {{- $jpegSet = $jpegSet | append (printf "%s %dw" $jpeg.RelPermalink $image.Width) -}} {{- end -}} {{- else -}} {{- if not $isStatic -}} {{- warnf "[render-image WARNING] Image resource not found for '%s' on page '%s'. Using fallback for external images or images not found in resources." $src .Page.Path -}} {{- end -}} {{- end -}}
